// 3. 必加方法(放在 @Component struct Index 内)
@State huaweiLoginState: string = '未登录';

async loginWithHuaweiID(): Promise<void> {
  try {
    const loginRequest = new authentication.HuaweiIDProvider()
      .createAuthorizationWithHuaweiIDRequest();
    loginRequest.forceAuthorization = true;
    loginRequest.scopes = ['profile', 'quickLoginAnonymousPhone'];
    loginRequest.permissions = ['serviceauthcode'];
    loginRequest.state = util.generateRandomUUID();
    loginRequest.nonce = util.generateRandomUUID();
    loginRequest.idTokenSignAlgorithm = authentication.IdTokenSignAlgorithm.PS256;

    const context = this.getUIContext().getHostContext() as common.UIAbilityContext;
    const controller = new authentication.AuthenticationController(context);
    controller.executeRequest(loginRequest, (error: BusinessError, data) => {
      if (error) {
        this.huaweiLoginState = `登录失败: ${error.message}`;
        return;
      }
      const response = data as authentication.AuthorizationWithHuaweiIDResponse;
      if (response.state && loginRequest.state !== response.state) {
        this.huaweiLoginState = '状态验证失败';
        return;
      }
      const credential = response.data;
      // 4. 客户端永远拿不到真实手机号,只有匿名手机号 + authorizationCode
      // 必发到后端换真实手机号
      this.sendToBackend(credential);
    });
  } catch (error) {
    this.huaweiLoginState = `异常: ${error}`;
  }
}

async sendToBackend(credential: authentication.HuaweiIDCredential): Promise<void> {
  // 必加:发送到 V1.4 后端 /v1/auth/huawei-login
  // 必用 Authorization Bearer API Key
  const response = await fetch('https://tunnel.zhishe.top/v1/auth/huawei-login', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': 'Bearer mKgd4EVhF7A8Y9zk6unOyb2jI1NBLaTR'
    },
    body: JSON.stringify({
      authorizationCode: credential.authorizationCode,
      openId: credential.openID,
      unionId: credential.unionID,
      anonymousPhone: credential.extraInfo?.quickLoginAnonymousPhone
    })
  });
  const data = await response.json();
  this.huaweiLoginState = `登录成功: ${JSON.stringify(data)}`;
}

// 5. 必加 UI 按钮(放在 Column 内)
Button('华为账号一键登录')

# app/api/auth.py(必加新接口)
@router.post("/v1/auth/huawei-login")
async def huawei_login(request: HuaweiLoginRequest, db = Depends(get_db)):
    """华为一键登录 - 用 authorizationCode 换真实手机号"""
    HUAWEI_API = "https://account-api.cloud.huawei.com/oauth2/v6/quickLogin/getPhoneNumber"
    
    # 1. 用 authorizationCode 换真实手机号
    response = requests.post(
        HUAWEI_API,
        json={
            "code": request.authorizationCode,
            "clientId": HUAWEI_CLIENT_ID,  # AGC 后台 OAuth 2.0 客户端 ID(42 位)
            "clientSecret": HUAWEI_CLIENT_SECRET  # AGC 后台 OAuth 2.0 客户端密钥
        },
        headers={"Content-Type": "application/json;charset=UTF-8"}
    )
    
    if response.status_code != 200:
        raise HTTPException(500, "华为账号服务异常")
    
    data = response.json()
    phone_number = data.get("phoneNumber")
    if not phone_number:
        raise HTTPException(400, "获取手机号失败")
    
    # 2. 查找/创建用户(用 UnionID 关联)
    user = db.query(User).filter(User.union_id == request.unionId).first()
    is_new_user = False
    if not user:
        user = User(
            union_id=request.unionId,
            open_id=request.openId,
            phone=phone_number,
            login_type="huawei"
        )
        db.add(user)
        is_new_user = True
    else:
        user.phone = phone_number
        user.open_id = request.openId
    db.commit()
    
    # 3. 签发 V1.4 后端 JWT Token
    token = create_access_token(user_id=user.id, expires_in="7d")
    
    return {
        "success": True,
        "token": token,
        "isNewUser": is_new_user,
        "user": {
            "id": user.id,
            "phone": phone_number,
            "nickname": user.nickname
        }
    }
```

# app/api/payment.py(必加新接口)
@router.post("/v1/payment/create-order")
async def create_order(request: CreateOrderRequest, user = Depends(require_api_key)):
    """创建华为支付订单 - 客户端拿到 orderStr 后调起支付"""
    
    # 1. 商户客户端创建订单(必 User 业务订单号)
    out_trade_no = f"ZS{user.id}{int(time.time())}"
    
    # 2. 调华为支付预下单接口拿 prepayId
    # 必用商户私钥(仅服务端,严禁客户端)做签名
    # 必用 SM2 + 华为支付证书
    HUAWEI_PAY_API = "https://pay-api.cloud.huawei.com/v1/pay/preorder"
    
    response = requests.post(
        HUAWEI_PAY_API,
        json={
            "mercNo": HUAWEI_MERC_NO,  # 商户号
            "appId": HUAWEI_APP_ID,  # AppID
            "preOrderId": out_trade_no,  # 商户订单号
            "totalAmount": request.amount,  # 金额(分)
            "productDesc": "知设装修顾问 VIP 30 天",
            "callbackUrl": "https://tunnel.zhishe.top/v1/payment/callback"  # 回调 URL
        },
        headers={
            "Content-Type": "application/json",
            "PayMercAuth": HUAWEI_PAY_AUTH  # 商户鉴权
        }
    )
    
    data = response.json()
    prepay_id = data.get("prepayId")
    
    # 3. 构建 orderStr 返回客户端
    order_str = json.dumps({
        "app_id": HUAWEI_APP_ID,
        "merc_no": HUAWEI_MERC_NO,
        "prepay_id": prepay_id,
        "timestamp": str(int(time.time() * 1000)),
        "noncestr": uuid.uuid4().hex,
        "sign": data.get("sign"),  # 后端签名
        "auth_id": user.union_id  # 必用华为账号 UnionID 关联
    })
    
    # 4. 订单入库
    db.add(Order(
        out_trade_no=out_trade_no,
        user_id=user.id,
        amount=request.amount,
        status="pending",
        prepay_id=prepay_id
    ))
    db.commit()
    
    return {"orderStr": order_str, "outTradeNo": out_trade_no}

@router.post("/v1/payment/callback")
async def payment_callback(request: Request):
    """华为支付回调 - 必 3 秒内响应 + 必 SM2 验签 + 必幂等"""
    body = await request.body()
    
    # 1. SM2 验签(必严格按华为支付规则)
    if not verify_sm2_signature(body, HUAWEI_PAY_PUBLIC_KEY):
        raise HTTPException(401, "验签失败")
    
    # 2. 解析回调(排序拼接 + sign 字段排除)
    data = parse_callback(body)
    out_trade_no = data.get("out_trade_no")
    
    # 3. 幂等检查(必用唯一键 out_trade_no)
    order = db.query(Order).filter(Order.out_trade_no == out_trade_no).first()
    if order.status == "paid":
        return {"code": "0", "message": "success"}  # 幂等返回
    
    # 4. 验签后入库
    order.status = "paid" if data.get("result_code") == "SUCCESS" else "failed"
    order.paid_at = datetime.now()
    db.commit()
    
    # 5. 必 3 秒内响应
    return {"code": "0", "message": "success"}
```
