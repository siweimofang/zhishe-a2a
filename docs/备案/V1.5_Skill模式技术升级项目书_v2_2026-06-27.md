```typescript
// 沙箱实证:华为官方 2025-03 公开示例
import { insightIntent, InsightIntentExecutor } from '@kit.AbilityKit';
import { window } from '@kit.ArkUI';
import { BusinessError } from '@kit.BasicServicesKit';

export default class InsightIntentExecutorImpl extends InsightIntentExecutor {
  private static readonly QUOTE = 'SearchRenovationQuote';
  private static readonly DESIGN = 'SearchRenovationDesign';
  private static readonly MATERIAL = 'SearchRenovationMaterial';
  private static readonly PITFALL = 'SearchRenovationPitfall';
  private static readonly LOCAL = 'SearchLocalRenovation';

  override onExecuteInUIAbilityForegroundMode(
    intentName: string,
    param: Record<string, Object>,
    pageLoader: window.WindowStage
  ): Promise<insightIntent.ExecuteResult> {
    switch (intentName) {
      case InsightIntentExecutorImpl.QUOTE:
        return this.handleQuote(param, pageLoader);
      case InsightIntentExecutorImpl.DESIGN:
        return this.handleDesign(param, pageLoader);
      // ... 其他 3 个意图
      default:
        return Promise.resolve({ code: -1, result: { message: 'unknown intent' } });
    }
  }

  private handleQuote(param: Record<string, Object>, pageLoader: window.WindowStage): Promise<insightIntent.ExecuteResult> {
    return new Promise((resolve) => {
      // 1. 从意图参数提取
      const city = param.city as string;
      const area = param.area as number;
      // 2. 路由到 Skill
      // 3. 拉起装修报价页面 + 预填
      // 4. 调用后端 API
      pageLoader.loadContent('pages/QuotePage', new LocalStorage({
        city, area, intentId: 'zhishe.renovation.quote'
      })).then(() => resolve({ code: 0, result: { message: 'ok' } }));
    });
  }
}
```
